cmake_minimum_required(VERSION 3.24)

project(NaturalehiaFaunaPackageConsumer LANGUAGES CXX)

find_package(NaturalehiaFauna 0.1 CONFIG REQUIRED)

add_executable(naturalehia_fauna_package_consumer main.cpp)
target_link_libraries(naturalehia_fauna_package_consumer PRIVATE Naturalehia::Fauna)
target_compile_features(naturalehia_fauna_package_consumer PRIVATE cxx_std_20)

enable_testing()
add_test(NAME naturalehia_fauna_package_consumer COMMAND naturalehia_fauna_package_consumer)
